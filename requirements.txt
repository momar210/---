scipy>=1.10,<2.0

# ----------------------------------------------------------------------------
# Image processing
# ----------------------------------------------------------------------------
# Used by: src/analysis/vectorize.py, src/parsers/docx_parser.py,
#          scripts/generate_diagrams.py (grayscale spectrum panel)
#
# Install note:
#   opencv-python is the standard build. If you run on a headless server,
#   you may instead use `opencv-python-headless` (same API, no GUI deps).
opencv-python>=4.8,<5.0

# Pillow is a transitive dependency of matplotlib and python-docx for
# image I/O. Pinning it explicitly avoids version drift across platforms.
Pillow>=10.0,<11.0

# ----------------------------------------------------------------------------
# Database / ORM
# ----------------------------------------------------------------------------
# Used by: src/database/connection.py, src/database/models.py,
#          src/database/queries.py, src/analysis/compare.py,
#          scripts/build_reference_database.py, scripts/compare_metrics.py
#
# SQLAlchemy 2.0 is required for the `select()` / `session.scalars()`
# patterns used throughout the enhanced modules.
SQLAlchemy>=2.0,<3.0

# ----------------------------------------------------------------------------
# DOCX parsing
# ----------------------------------------------------------------------------
# Used by: src/parsers/docx_parser.py, scripts/generate_diagrams.py
#
# python-docx reads .docx files, exposes embedded images via
# document.part.rels, and is the source of the RELATIONSHIP_TYPE constant
# used for image detection.

# ----------------------------------------------------------------------------
# Plotting / visualization
# ----------------------------------------------------------------------------
# Used by: scripts/generate_diagrams.py,
